% ------------------------------------------------------------------ numbering
% Chapters are numbered with Roman numerals (I--VI), as in the original.
\renewcommand{\thechapter}{\Roman{chapter}.}
% Sections carry a trailing point, as in the original table of contents.
\renewcommand{\thesection}{\arabic{section}}

% ------------------------------------------------------------------ theorems
% Gabriel does NOT use one running counter.  Verified against the scan:
%   * propositions run continuously across the SECTIONS of a chapter, and
%     restart at each chapter   (Ch. I ends at Proposition 15 on p. 345;
%     Ch. II opens at Proposition 1 on p. 346);
%   * theorems restart at each chapter;
%   * lemmas restart at each SECTION (Ch. II §2 opens with
%     "Lemme 1" on p. 349, after §1 has run to "Lemme 4" on p. 348);
%   * corollaries are numbered afresh after each PROPOSITION
%     (Ch. V §6 prints Corollaries 1 and 2 to Proposition 11, then
%     Corollaries 1, 2 and 3 to Proposition 13, then Corollaries 1
%     and 2 to Proposition 14; Gabriel cites them as "corollaire N
%     de la proposition M" throughout);
%   * and some corollaries are printed with no number at all ("Corollaire."
%     on p. 360, between Proposition 9 and Proposition 10).
% The counters below reproduce that exactly; do not "simplify" them.
% NOTE ON SYNTAX.  With amsthm loaded the counter must be created by
% \newtheorem itself.  Declaring \newcounter{prop} and then writing
% \newtheorem{proposition}{Proposition}[prop] makes amsthm print the number as
% "<parent>.<counter>", i.e. "Proposition 0.1."  So: declare the environments
% without the optional [counter] argument, and reset them afterwards.
\newtheorem{theorem}{Theorem}
\newtheorem{proposition}{Proposition}
\newtheorem{lemma}{Lemma}
\newtheorem{corollary}{Corollary}
\theoremstyle{definition}
\newtheorem{definition}{Definition}
\newtheorem{remark}{Remark}

% For the corollaries Gabriel states without a number.
\newtheorem*{unnumberedcorollary}{Corollary}

% The resets themselves.  These must come AFTER the \newtheorem calls,
% so that the counters exist by now.
\makeatletter
\@addtoreset{theorem}{chapter}
\@addtoreset{proposition}{chapter}
\@addtoreset{lemma}{section}
\@addtoreset{corollary}{section}
% Gabriel numbers the corollaries of each proposition (or theorem) afresh:
% the corollary counter must therefore also restart at each proposition.
% Verified 2026-10-16 against Ch. V §6 (Cor. 1-2 to Prop. 11, Cor. 1-3 to
% Prop. 13, Cor. 1-2 to Prop. 14): this changes nothing in any section
% whose numbered corollaries all follow a single proposition, which is
% the case of every section translated so far.
\@addtoreset{corollary}{proposition}
\@addtoreset{definition}{section}
\@addtoreset{remark}{section}
\@addtoreset{footnote}{chapter}
\makeatother

% ------------------------------------------------------------------ headings
% The original prints section headings in bold, numbered, run in with a dash
% into the first paragraph.  emulating that keeps the running text close to
% the original, which matters when reading the two side by side.

\titlespacing*{\section}{0pt}{1.2\baselineskip}{0.6\baselineskip}
